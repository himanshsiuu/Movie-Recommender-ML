import torch
import torch.nn as nn
import torch.nn.functional as F

class TwoTowerRecommender(nn.Module):
    def __init__(self, num_users, num_items, num_genres=15, embed_dim=64):
        super(TwoTowerRecommender, self).__init__()
        # User Tower
        self.user_embedding = nn.Embedding(num_users, embed_dim)
        self.user_fc = nn.Sequential(
            nn.Linear(embed_dim, 128),
            nn.BatchNorm1d(128),
            nn.ReLU(),
            nn.Dropout(0.1),
            nn.Linear(128, embed_dim)
        )
        # Item Tower
        self.item_embedding = nn.Embedding(num_items, embed_dim)
        self.genre_linear = nn.Linear(num_genres, embed_dim)
        self.item_fc = nn.Sequential(
            nn.Linear(embed_dim * 2, 128),
            nn.BatchNorm1d(128),
            nn.ReLU(),
            nn.Dropout(0.1),
            nn.Linear(128, embed_dim)
        )
        self.temperature = nn.Parameter(torch.ones(1) * 0.07)

    def forward(self, user_ids, item_ids, item_genres):
        u_raw = self.user_embedding(user_ids)
        u_vec = F.normalize(self.user_fc(u_raw), p=2, dim=-1)
        
        i_raw = self.item_embedding(item_ids)
        g_raw = self.genre_linear(item_genres)
        combined_item = torch.cat([i_raw, g_raw], dim=-1)
        v_vec = F.normalize(self.item_fc(combined_item), p=2, dim=-1)
        
        logits = torch.matmul(u_vec, v_vec.T) / self.temperature
        return logits
